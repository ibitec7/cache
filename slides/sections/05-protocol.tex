\section{Protocol}

\begin{frame}{Split by time, not at random}
  \begin{itemize}
    \item Fit, validation and held-out blocks are contiguous in request order.
    \item No interleaving, no random shuffle.
    \item Reuse distances are computed once over the whole window, so a
      distance crossing a split boundary is never truncated.
    \item Ground truth is the empirical survival on each block separately.
  \end{itemize}
\end{frame}

\begin{frame}{Fixed before fitting}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.52\textwidth}
      \textbf{Capacity grid}
      \begin{itemize}
        \item Upper edges of the hazard bins, at most 24 points, all below the
          overflow edge.
        \item Same grid for the truth and for every model.
      \end{itemize}
      \vspace{0.5em}
      \textbf{Hyperparameters}
      \begin{itemize}
        \item Optuna, tree Parzen estimator.
        \item Objective: validation error of the curve.
        \item Headline uses defaults, so every row shares one protocol.
      \end{itemize}
    \end{column}
    \begin{column}{0.44\textwidth}
      \begin{block}{Cluster selection}
        First five million requests of each shard, fixed before any fit.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}
